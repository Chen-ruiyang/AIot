void setup() {
    Serial.begin(115200);

void loop() {
    float d = Get_Distance();
    if (d < 30.0) { 
        Emergency_Stop();
        Execute_Reroute(); 
    } else {
        Keep_Speed(1.2); 
    }
    delay(10);
}
